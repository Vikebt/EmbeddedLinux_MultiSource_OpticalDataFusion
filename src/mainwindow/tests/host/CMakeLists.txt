cmake_minimum_required(VERSION 3.10)
project(mainwindow_host_tests CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(test_reconnect_backoff test_reconnect_backoff.cpp)
target_include_directories(test_reconnect_backoff PRIVATE ../../include)
enable_testing()
add_test(NAME reconnect_backoff COMMAND test_reconnect_backoff)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # WSL may inherit Windows/Conda CMake package paths containing .lib files.
    find_package(yaml-cpp QUIET CONFIG PATHS /usr /usr/local NO_DEFAULT_PATH)
else()
    find_package(yaml-cpp QUIET CONFIG)
endif()
if(TARGET yaml-cpp)
    add_executable(test_yaml_config test_yaml_config.cpp ../../src/yaml_config.cpp)
    target_include_directories(test_yaml_config PRIVATE ../../include)
    target_link_libraries(test_yaml_config PRIVATE yaml-cpp)
elseif(TARGET yaml-cpp::yaml-cpp)
    add_executable(test_yaml_config test_yaml_config.cpp ../../src/yaml_config.cpp)
    target_include_directories(test_yaml_config PRIVATE ../../include)
    target_link_libraries(test_yaml_config PRIVATE yaml-cpp::yaml-cpp)
else()
    message(STATUS "yaml-cpp unavailable: yaml_config_transaction test not built")
endif()

if(TARGET test_yaml_config)
    if(MSVC)
        # Installed Conda yaml-cpp is a Release-only C++ DLL. A matching
        # Debug DLL is needed before this test can run with MSVC Debug STL.
        add_test(NAME yaml_config_transaction CONFIGURATIONS Release COMMAND test_yaml_config)
        message(STATUS "MSVC Debug YAML test unavailable: matching Debug yaml-cpp DLL not installed")
    else()
        add_test(NAME yaml_config_transaction COMMAND test_yaml_config)
    endif()
endif()
